    <script>
        // 单选切换逻辑
        document.querySelectorAll('.radio-dot').forEach(radioEl => {
            const label = radioEl.closest('label');
            label.addEventListener('click', function() {
                const groupName = this.querySelector('.dropdown-trigger').textContent.trim();
                const parentRow = this.closest('.flex');
                parentRow.querySelectorAll('.radio-dot').forEach(dot => {
                    dot.innerHTML = '';
                    dot.classList.remove('border-primary');
                    dot.classList.add('border-gray-400');
                });
                radioEl.classList.remove('border-gray-400');
                radioEl.classList.add('border-primary');
                radioEl.innerHTML = '<span class="radio-dot-inner"></span>';
            })
        })

        // 复选框切换逻辑
        document.querySelectorAll('.checkbox-box').forEach(checkboxEl => {
            const label = checkboxEl.closest('label');
            label.addEventListener('click', function() {
                if(checkboxEl.classList.contains('bg-primary')) {
                    checkboxEl.classList.remove('bg-primary', 'text-white');
                    checkboxEl.innerHTML = '';
                } else {
                    checkboxEl.classList.add('bg-primary', 'text-white');
                    checkboxEl.innerHTML = '<i class="fa fa-check text-xs"></i>';
                }
            })
        })

        // 删除已选标签
        document.querySelectorAll('.tag-item button').forEach(closeBtn => {
            closeBtn.addEventListener('click', function(e) {
                e.stopPropagation();
                this.closest('.tag-item').remove();
            })
        })
    </script>
